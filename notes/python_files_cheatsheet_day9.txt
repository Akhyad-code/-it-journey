open("file.txt", "w") — создать или перезаписать файл
open("file.txt", "r") — прочитать файл
open("file.txt", "a") — добавить текст в конец файла
file.write("text") — записать текст
file.read() — прочитать текст
import os — подключить работу с системой
os.listdir(".") — показать файлы в папке
os.path.exists("name") — проверить, существует ли файл или папка
os.mkdir("name") — создать папку
